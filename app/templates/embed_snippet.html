<form
  hx-post="{{ request.base_url }}form/{{ forms[0]['slug'] }}"
  hx-target="#form-response"
  hx-swap="innerHTML"
  action="{{ request.base_url }}form/{{ forms[0]['slug'] }}"
  method="post"
>
{% for field in (forms[0].fields or []) %}
   {% if field.type == 'textarea' %}
   <textarea name="{{ field.name }}" placeholder="{{ field.label }}"{% if field.required %} required{% endif %} rows="5"></textarea>
   {% elif field.type == 'select' %}
   <select name="{{ field.name }}"{% if field.required %} required{% endif %}>
     <option value="">Select {{ field.label }}</option>
     {% for opt in field.options %}
     <option value="{{ opt }}">{{ opt }}</option>
     {% endfor %}
   </select>
   {% elif field.type == 'checkbox' %}
   <label><input type="checkbox" name="{{ field.name }}" value="yes" /> {{ field.label }}</label>
   {% else %}
   <input type="{{ field.type }}" name="{{ field.name }}" placeholder="{{ field.label }}"{% if field.required %} required{% endif %}{% if field.type == 'text' %} maxlength="500"{% elif field.type == 'email' %} maxlength="254"{% elif field.type == 'tel' %} maxlength="20"{% elif field.type == 'url' %} maxlength="2000"{% elif field.type == 'number' %} min="0" max="999999999"{% endif %} />
   {% endif %}
{% endfor %}
  <script src="https://cdn.jsdelivr.net/npm/cap-widget"></script>
  <cap-widget required
    data-cap-api-endpoint="{{ request.base_url }}cap-proxy/{% if settings.cap_site_key %}{{ settings.cap_site_key }}/{% endif %}">
  </cap-widget>
  <button type="submit">Send</button>
</form>
<div id="form-response"></div>
